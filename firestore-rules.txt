rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Users collection - user chỉ có thể đọc/ghi profile của mình
    match /users/{userId} {
      allow read, write, create, update: if request.auth != null && request.auth.uid == userId;
    }
    
    // Usage tracking collection - user chỉ có thể truy cập thời gian sử dụng của mình
    match /usage/{userId} {
      allow read, write, create, update: if request.auth != null && request.auth.uid == userId;
      
      // Daily usage subcollection
      match /daily/{date} {
        allow read, write, create, update: if request.auth != null && request.auth.uid == userId;
      }
    }
    
    // Study time collection - user chỉ có thể truy cập thời gian học của mình (legacy)
    match /studytime/{userId} {
      allow read, write, create, update: if request.auth != null && request.auth.uid == userId;
    }
  }
}
